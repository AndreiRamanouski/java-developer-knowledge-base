# MySQL Replication and Read Replicas with Spring Boot

## Deep Dive Interview Guide

This guide covers MySQL replication strategies for production Spring Boot applications, including master-slave setup, read/write splitting, lag handling, and failover.

**Interview Tip**: Replication questions test understanding of distributed systems, consistency trade-offs, performance optimization, and production operational concerns.

---

## Q3: Replication and Read Replicas with Spring Boot

**Short Answer**: Master-slave replication: Master handles writes, slaves handle reads. Configure multiple datasources in Spring Boot. Use @Transactional(readOnly=true) to route reads to slaves. Implement AbstractRoutingDataSource with ThreadLocal to determine routing. Handle replication lag with "read-your-writes" consistency using master for recent updates. Failover: Promote slave to master, use orchestration tools (Orchestrator, MHA). Monitor with replication delay metrics (Seconds_Behind_Master).

## 1. Master-Slave Configuration

### MySQL Replication Setup

```sql
-- === MASTER Configuration ===

-- Edit /etc/mysql/my.cnf (or my.ini on Windows)
[mysqld]
server-id = 1                          # Unique ID for master
log-bin = /var/log/mysql/mysql-bin     # Enable binary logging
binlog-format = ROW                    # ROW format (recommended)
binlog-do-db = mydb                    # Replicate specific database (optional)
expire_logs_days = 7                   # Keep binary logs for 7 days

# Restart MySQL
sudo systemctl restart mysql

-- Create replication user
CREATE USER 'replication_user'@'%' IDENTIFIED BY 'strong_password';
GRANT REPLICATION SLAVE ON *.* TO 'replication_user'@'%';
FLUSH PRIVILEGES;

-- Get master status (note File and Position)
SHOW MASTER STATUS;
-- Result:
-- +------------------+----------+--------------+------------------+
-- | File             | Position | Binlog_Do_DB | Binlog_Ignore_DB |
-- +------------------+----------+--------------+------------------+
-- | mysql-bin.000001 |      154 | mydb         |                  |
-- +------------------+----------+--------------+------------------+


-- === SLAVE Configuration ===

-- Edit /etc/mysql/my.cnf
[mysqld]
server-id = 2                          # Unique ID (different from master)
relay-log = /var/log/mysql/relay-bin   # Enable relay logs
read-only = 1                          # Make slave read-only
log-slave-updates = 1                  # Log replicated events (for chaining)

# Restart MySQL
sudo systemctl restart mysql

-- Configure slave to connect to master
CHANGE MASTER TO
    MASTER_HOST='master.example.com',
    MASTER_USER='replication_user',
    MASTER_PASSWORD='strong_password',
    MASTER_LOG_FILE='mysql-bin.000001',  -- From SHOW MASTER STATUS
    MASTER_LOG_POS=154;                  -- From SHOW MASTER STATUS

-- Start replication
START SLAVE;

-- Check slave status
SHOW SLAVE STATUS\G

-- Key fields to check:
-- Slave_IO_Running: Yes (connected to master)
-- Slave_SQL_Running: Yes (applying changes)
-- Seconds_Behind_Master: 0 (no lag)
-- Last_Error: (should be empty)


-- === Multiple Slaves ===

-- Slave 2
server-id = 3
-- Same configuration as Slave 1

-- Slave 3
server-id = 4
-- Same configuration as Slave 1

-- Master sends changes to all slaves simultaneously
```

### DataSource Configuration

```java
@Configuration
public class DataSourceConfig {
    
    @Bean
    @ConfigurationProperties("spring.datasource.master")
    public DataSource masterDataSource() {
        return DataSourceBuilder.create()
                .type(HikariDataSource.class)
                .build();
    }
    
    @Bean
    @ConfigurationProperties("spring.datasource.slave1")
    public DataSource slave1DataSource() {
        return DataSourceBuilder.create()
                .type(HikariDataSource.class)
                .build();
    }
    
    @Bean
    @ConfigurationProperties("spring.datasource.slave2")
    public DataSource slave2DataSource() {
        return DataSourceBuilder.create()
                .type(HikariDataSource.class)
                .build();
    }
    
    @Bean
    public DataSource routingDataSource(
            @Qualifier("masterDataSource") DataSource master,
            @Qualifier("slave1DataSource") DataSource slave1,
            @Qualifier("slave2DataSource") DataSource slave2) {
        
        ReplicationRoutingDataSource routingDataSource = new ReplicationRoutingDataSource();
        
        // Set default datasource
        routingDataSource.setDefaultTargetDataSource(master);
        
        // Set target datasources
        Map<Object, Object> dataSourceMap = new HashMap<>();
        dataSourceMap.put(DataSourceType.MASTER, master);
        dataSourceMap.put(DataSourceType.SLAVE, slave1);  // First slave
        dataSourceMap.put(DataSourceType.SLAVE2, slave2); // Second slave
        routingDataSource.setTargetDataSources(dataSourceMap);
        
        return routingDataSource;
    }
    
    @Bean
    @Primary
    public DataSource dataSource(
            @Qualifier("routingDataSource") DataSource routingDataSource) {
        return new LazyConnectionDataSourceProxy(routingDataSource);
        // LazyConnectionDataSourceProxy delays connection until actual query
        // This allows transaction context to be set before connection
    }
}

enum DataSourceType {
    MASTER, SLAVE, SLAVE2
}
```

### Using @Transactional(readOnly = true)

```java
@Service
public class UserService {
    
    @Autowired
    private UserRepository userRepository;
    
    // WRITE operations → Master
    @Transactional  // readOnly = false (default)
    public User createUser(User user) {
        return userRepository.save(user);
        // Routes to MASTER datasource
    }
    
    @Transactional
    public User updateUser(Long id, String name) {
        User user = userRepository.findById(id).orElseThrow();
        user.setName(name);
        return userRepository.save(user);
        // Routes to MASTER datasource
    }
    
    @Transactional
    public void deleteUser(Long id) {
        userRepository.deleteById(id);
        // Routes to MASTER datasource
    }
    
    // READ operations → Slave
    @Transactional(readOnly = true)
    public User findById(Long id) {
        return userRepository.findById(id).orElseThrow();
        // Routes to SLAVE datasource
    }
    
    @Transactional(readOnly = true)
    public List<User> findAll() {
        return userRepository.findAll();
        // Routes to SLAVE datasource
    }
    
    @Transactional(readOnly = true)
    public Page<User> findByCity(String city, Pageable pageable) {
        return userRepository.findByCity(city, pageable);
        // Routes to SLAVE datasource
    }
    
    // READ after WRITE → Master (avoid replication lag)
    @Transactional  // readOnly = false
    public User createAndReturn(User user) {
        User saved = userRepository.save(user);
        // Subsequent read in same transaction uses MASTER
        return userRepository.findById(saved.getId()).orElseThrow();
    }
}
```

## 3. AbstractRoutingDataSource for Read/Write Splitting

### Custom Routing DataSource

```java
public class ReplicationRoutingDataSource extends AbstractRoutingDataSource {
    
    private static final ThreadLocal<DataSourceType> contextHolder = 
        new ThreadLocal<>();
    
    private final List<DataSourceType> slaves = Arrays.asList(
        DataSourceType.SLAVE, 
        DataSourceType.SLAVE2
    );
    
    private final AtomicInteger slaveIndex = new AtomicInteger(0);
    
    public static void setDataSourceType(DataSourceType dataSourceType) {
        contextHolder.set(dataSourceType);
    }
    
    public static DataSourceType getDataSourceType() {
        return contextHolder.get();
    }
    
    public static void clearDataSourceType() {
        contextHolder.remove();
    }
    
    @Override
    protected Object determineCurrentLookupKey() {
        DataSourceType type = getDataSourceType();
        
        if (type == null) {
            // Default to master if not specified
            return DataSourceType.MASTER;
        }
        
        if (type == DataSourceType.SLAVE) {
            // Round-robin load balancing between slaves
            return roundRobinSlave();
        }
        
        return type;
    }
    
    private DataSourceType roundRobinSlave() {
        int index = Math.abs(slaveIndex.getAndIncrement() % slaves.size());
        return slaves.get(index);
    }
    
    @Override
    public void afterPropertiesSet() {
        super.afterPropertiesSet();
        // Verify all slaves are accessible
        verifySlaveConnections();
    }
    
    private void verifySlaveConnections() {
        Map<Object, DataSource> resolvedDataSources = 
            (Map<Object, DataSource>) super.getResolvedDataSources();
        
        for (DataSourceType slave : slaves) {
            DataSource ds = resolvedDataSources.get(slave);
            if (ds != null) {
                try (Connection conn = ds.getConnection()) {
                    log.info("Slave {} is accessible", slave);
                } catch (SQLException e) {
                    log.error("Slave {} is not accessible", slave, e);
                }
            }
        }
    }
}
```

### Load Balancing Strategies

```java
// === Strategy 1: Round-Robin (Implemented Above) ===
// Distributes load evenly across all slaves

// === Strategy 2: Weighted Round-Robin ===
public class WeightedRoundRobinStrategy {
    
    private final List<WeightedDataSource> slaves;
    private final AtomicInteger index = new AtomicInteger(0);
    
    public WeightedRoundRobinStrategy() {
        this.slaves = Arrays.asList(
            new WeightedDataSource(DataSourceType.SLAVE, 70),   // 70% traffic
            new WeightedDataSource(DataSourceType.SLAVE2, 30)   // 30% traffic
        );
    }
    
    public DataSourceType selectSlave() {
        int total = slaves.stream().mapToInt(w -> w.weight).sum();
        int random = ThreadLocalRandom.current().nextInt(total);
        
        int cumulative = 0;
        for (WeightedDataSource slave : slaves) {
            cumulative += slave.weight;
            if (random < cumulative) {
                return slave.dataSource;
            }
        }
        
        return slaves.get(0).dataSource;  // Fallback
    }
    
    @Data
    @AllArgsConstructor
    static class WeightedDataSource {
        DataSourceType dataSource;
        int weight;
    }
}

// === Strategy 3: Least Connections ===
public class LeastConnectionsStrategy {
    
    private final Map<DataSourceType, AtomicInteger> activeConnections;
    
    public LeastConnectionsStrategy() {
        this.activeConnections = new ConcurrentHashMap<>();
        activeConnections.put(DataSourceType.SLAVE, new AtomicInteger(0));
        activeConnections.put(DataSourceType.SLAVE2, new AtomicInteger(0));
    }
    
    public DataSourceType selectSlave() {
        return activeConnections.entrySet().stream()
            .min(Comparator.comparingInt(e -> e.getValue().get()))
            .map(Map.Entry::getKey)
            .orElse(DataSourceType.SLAVE);
    }
    
    public void incrementConnections(DataSourceType type) {
        activeConnections.get(type).incrementAndGet();
    }
    
    public void decrementConnections(DataSourceType type) {
        activeConnections.get(type).decrementAndGet();
    }
}

// === Strategy 4: Random ===
public class RandomStrategy {
    
    private final List<DataSourceType> slaves = Arrays.asList(
        DataSourceType.SLAVE,
        DataSourceType.SLAVE2
    );
    
    public DataSourceType selectSlave() {
        int index = ThreadLocalRandom.current().nextInt(slaves.size());
        return slaves.get(index);
    }
}
```

### Force Master for Critical Operations

```java
@Service
public class OrderService {
    
    @Autowired
    private OrderRepository orderRepository;
    
    @Autowired
    private UserRepository userRepository;
    
    // Normal read → Slave
    @Transactional(readOnly = true)
    public Order findById(Long id) {
        return orderRepository.findById(id).orElseThrow();
    }
    
    // Force master for "read-your-writes" consistency
    @Transactional  // readOnly = false → Master
    public Order getMyRecentOrder(Long userId) {
        // User just placed order, must read from master to guarantee visibility
        return orderRepository.findTopByUserIdOrderByCreatedAtDesc(userId)
            .orElseThrow();
    }
    
    // Write then read → Same transaction (Master)
    @Transactional
    public OrderResponse createOrder(OrderRequest request) {
        // Write
        Order order = new Order();
        order.setUserId(request.getUserId());
        order.setTotal(request.getTotal());
        Order saved = orderRepository.save(order);
        
        // Read in same transaction (uses Master, no lag)
        User user = userRepository.findById(request.getUserId()).orElseThrow();
        
        return new OrderResponse(saved, user);
    }
    
    // Manual routing to master
    @Transactional(readOnly = true)
    public Order findByIdFromMaster(Long id) {
        // Override routing
        ReplicationRoutingDataSource.setDataSourceType(DataSourceType.MASTER);
        try {
            return orderRepository.findById(id).orElseThrow();
        } finally {
            ReplicationRoutingDataSource.clearDataSourceType();
        }
    }
}
```

## 4. Handling Replication Lag

### Understanding Replication Lag

```sql
-- Check replication lag on slave
SHOW SLAVE STATUS\G

-- Key field: Seconds_Behind_Master
-- 0 = No lag (slave is caught up)
-- 5 = Slave is 5 seconds behind
-- NULL = Replication not running

-- Causes of replication lag:
-- 1. High write load on master
-- 2. Slow slave hardware
-- 3. Network latency
-- 4. Large transactions
-- 5. Lock contention on slave
```

### Lag Monitoring Service

```java
@Service
public class ReplicationLagMonitor {
    
    @Autowired
    private JdbcTemplate slaveJdbcTemplate;
    
    @Autowired
    private MeterRegistry meterRegistry;
    
    private final AtomicLong currentLag = new AtomicLong(0);
    
    @PostConstruct
    public void init() {
        // Register gauge metric
        meterRegistry.gauge("mysql.replication.lag.seconds", currentLag);
    }
    
    @Scheduled(fixedDelay = 5000)  // Check every 5 seconds
    public void checkReplicationLag() {
        try {
            String sql = "SHOW SLAVE STATUS";
            
            Map<String, Object> status = slaveJdbcTemplate.queryForMap(sql);
            
            Object secondsBehind = status.get("Seconds_Behind_Master");
            
            if (secondsBehind == null) {
                log.error("Replication is not running!");
                currentLag.set(-1);  // -1 indicates error
                // Alert operations team
                alertReplicationStopped();
            } else {
                long lag = ((Number) secondsBehind).longValue();
                currentLag.set(lag);
                
                if (lag > 10) {
                    log.warn("High replication lag: {} seconds", lag);
                    // Send alert if lag > 30 seconds
                    if (lag > 30) {
                        alertHighReplicationLag(lag);
                    }
                }
            }
            
        } catch (Exception e) {
            log.error("Failed to check replication lag", e);
        }
    }
    
    public long getCurrentLag() {
        return currentLag.get();
    }
    
    public boolean isLagAcceptable(long maxLagSeconds) {
        long lag = currentLag.get();
        return lag >= 0 && lag <= maxLagSeconds;
    }
    
    private void alertReplicationStopped() {
        // Send alert via email, Slack, PagerDuty, etc.
        log.error("ALERT: Replication has stopped!");
    }
    
    private void alertHighReplicationLag(long lag) {
        // Send warning
        log.warn("ALERT: High replication lag: {} seconds", lag);
    }
}
```

### Lag-Aware Routing

```java
@Component
public class LagAwareRoutingDataSource extends ReplicationRoutingDataSource {
    
    @Autowired
    private ReplicationLagMonitor lagMonitor;
    
    private static final long MAX_ACCEPTABLE_LAG_SECONDS = 5;
    
    @Override
    protected Object determineCurrentLookupKey() {
        DataSourceType type = getDataSourceType();
        
        if (type == DataSourceType.SLAVE) {
            // Check if lag is acceptable
            if (!lagMonitor.isLagAcceptable(MAX_ACCEPTABLE_LAG_SECONDS)) {
                log.warn("Replication lag too high, routing to MASTER");
                return DataSourceType.MASTER;  // Fallback to master
            }
            
            return roundRobinSlave();
        }
        
        return type != null ? type : DataSourceType.MASTER;
    }
}
```

### Read-Your-Writes Consistency

```java
@Service
public class ConsistentReadService {
    
    @Autowired
    private ReplicationLagMonitor lagMonitor;
    
    // Store timestamp of last write per user
    private final Cache<Long, Instant> lastWriteTimestamp = 
        Caffeine.newBuilder()
            .maximumSize(10000)
            .expireAfterWrite(Duration.ofMinutes(5))
            .build();
    
    // Write operation
    @Transactional
    public User createUser(User user) {
        User saved = userRepository.save(user);
        
        // Record write timestamp
        lastWriteTimestamp.put(saved.getId(), Instant.now());
        
        return saved;
    }
    
    // Read with consistency guarantee
    @Transactional(readOnly = true)
    public User findUserConsistent(Long userId) {
        Instant lastWrite = lastWriteTimestamp.getIfPresent(userId);
        
        if (lastWrite != null) {
            // Recent write - check if safe to read from slave
            Duration timeSinceWrite = Duration.between(lastWrite, Instant.now());
            long lagSeconds = lagMonitor.getCurrentLag();
            
            if (timeSinceWrite.getSeconds() < lagSeconds + 1) {
                // Too risky - data might not be on slave yet
                // Force read from master
                ReplicationRoutingDataSource.setDataSourceType(DataSourceType.MASTER);
                log.info("Reading from MASTER for consistency (recent write)");
            }
        }
        
        return userRepository.findById(userId).orElseThrow();
    }
}
```

```java
@Component
public class SessionConsistencyManager {
    
    // Track which users have recent writes
    private final Set<Long> recentWriters = ConcurrentHashMap.newKeySet();
    
    @Scheduled(fixedDelay = 60000)  // Clean up every minute
    public void cleanupRecentWriters() {
        recentWriters.clear();
    }
    
    public void markRecentWrite(Long userId) {
        recentWriters.add(userId);
    }
    
    public boolean hasRecentWrite(Long userId) {
        return recentWriters.contains(userId);
    }
}
```

## 5. Failover Strategies

### Understanding Failover

```
Failover Scenarios:

1. Master Failure:
   - Promote slave to master
   - Reconfigure remaining slaves
   - Update application configuration

2. Slave Failure:
   - Remove from pool
   - Route traffic to remaining slaves
   - Investigate and fix

3. Network Partition:
   - Detect split-brain
   - Choose primary side
   - Prevent data conflicts
```

### Manual Failover Process

```sql
-- === When Master Fails ===

-- Step 1: Stop writes on all slaves
STOP SLAVE;

-- Step 2: Check which slave is most up-to-date
SHOW SLAVE STATUS\G
-- Look at: Master_Log_File, Read_Master_Log_Pos, Exec_Master_Log_Pos

-- Choose slave with highest position

-- Step 3: Promote chosen slave to master
-- On the new master (was slave1):
STOP SLAVE;
RESET SLAVE ALL;
SET GLOBAL read_only = 0;  -- Make writable

-- Enable binary logging (if not already enabled)
SET GLOBAL log_bin = ON;

-- Step 4: Point remaining slaves to new master
-- On slave2:
STOP SLAVE;

CHANGE MASTER TO
    MASTER_HOST='new-master.example.com',
    MASTER_USER='replication_user',
    MASTER_PASSWORD='strong_password',
    MASTER_LOG_FILE='mysql-bin.000001',  -- From new master
    MASTER_LOG_POS=154;

START SLAVE;

-- Step 5: Verify replication
SHOW SLAVE STATUS\G
-- Slave_IO_Running: Yes
-- Slave_SQL_Running: Yes
```

```java
            @Qualifier("masterDataSource") DataSource master,
            @Qualifier("slave1DataSource") DataSource slave1,
            @Qualifier("slave2DataSource") DataSource slave2) {
        
        FailoverRoutingDataSource dataSource = new FailoverRoutingDataSource();
        dataSource.setMasterDataSource(master);
        dataSource.setSlaveDataSources(Arrays.asList(slave1, slave2));
        
        return dataSource;
    }
}

public class FailoverRoutingDataSource extends AbstractRoutingDataSource {
    
    private DataSource masterDataSource;
    private List<DataSource> slaveDataSources;
    private final Set<DataSource> unhealthyDataSources = ConcurrentHashMap.newKeySet();
    
    @Autowired
    private DataSourceHealthChecker healthChecker;
    
    public void setMasterDataSource(DataSource master) {
        this.masterDataSource = master;
    }
    
    public void setSlaveDataSources(List<DataSource> slaves) {
        this.slaveDataSources = slaves;
    }
    
    @Override
    protected Object determineCurrentLookupKey() {
        DataSourceType type = ReplicationRoutingDataSource.getDataSourceType();
        
        if (type == DataSourceType.MASTER) {
            return getHealthyMaster();
        } else if (type == DataSourceType.SLAVE) {
            return getHealthySlave();
        }
        
        return DataSourceType.MASTER;
    }
    
    private DataSource getHealthyMaster() {
        if (isHealthy(masterDataSource)) {
            return masterDataSource;
        }
        
        log.error("Master is unhealthy! Attempting to promote slave...");
        // In real scenario, this would trigger orchestrated failover
        // For now, just mark as unhealthy
        unhealthyDataSources.add(masterDataSource);
        
        throw new DataSourceConnectionException("Master database is unavailable");
    }
    
    private DataSource getHealthySlave() {
        List<DataSource> healthySlaves = slaveDataSources.stream()
            .filter(this::isHealthy)
            .collect(Collectors.toList());
        
        if (healthySlaves.isEmpty()) {
            log.warn("No healthy slaves available, falling back to master");
            return masterDataSource;
        }
        
        // Round-robin among healthy slaves
        int index = ThreadLocalRandom.current().nextInt(healthySlaves.size());
        return healthySlaves.get(index);
    }
    
    private boolean isHealthy(DataSource dataSource) {
        if (unhealthyDataSources.contains(dataSource)) {
            // Check if it's recovered
            if (healthChecker.check(dataSource)) {
                unhealthyDataSources.remove(dataSource);
                log.info("DataSource recovered: {}", dataSource);
                return true;
            }
            return false;
        }
        
        return healthChecker.check(dataSource);
    }
}

@Service
public class DataSourceHealthChecker {
    
    private final Map<DataSource, Instant> lastCheck = new ConcurrentHashMap<>();
    private final Map<DataSource, Boolean> healthStatus = new ConcurrentHashMap<>();
    
    private static final Duration CHECK_INTERVAL = Duration.ofSeconds(5);
    
    public boolean check(DataSource dataSource) {
        Instant lastCheckTime = lastCheck.get(dataSource);
        
        if (lastCheckTime != null && 
            Duration.between(lastCheckTime, Instant.now()).compareTo(CHECK_INTERVAL) < 0) {
            // Return cached result if checked recently
            return healthStatus.getOrDefault(dataSource, false);
        }
        
        boolean healthy = performHealthCheck(dataSource);
        lastCheck.put(dataSource, Instant.now());
        healthStatus.put(dataSource, healthy);
        
        return healthy;
    }
    
    private boolean performHealthCheck(DataSource dataSource) {
        try (Connection conn = dataSource.getConnection()) {
            // Simple ping query
            try (Statement stmt = conn.createStatement()) {
                stmt.execute("SELECT 1");
                return true;
            }
        } catch (SQLException e) {
            log.error("Health check failed for datasource", e);
            return false;
        }
    }
}
```

```java
@Service
public class DynamicDataSourceConfigService {
    
    @Autowired
    private Environment environment;
    
    @Autowired
    private ApplicationContext applicationContext;
    
    // Update master datasource after failover
    public void updateMasterDataSource(String newMasterHost) {
        log.info("Updating master datasource to: {}", newMasterHost);
        
        // 1. Create new datasource
        HikariDataSource newMaster = new HikariDataSource();
        newMaster.setJdbcUrl(String.format(
            "jdbc:mysql://%s:3306/mydb", 
            newMasterHost
        ));
        newMaster.setUsername(environment.getProperty("spring.datasource.master.username"));
        newMaster.setPassword(environment.getProperty("spring.datasource.master.password"));
        
        // 2. Get routing datasource
        ReplicationRoutingDataSource routingDs = 
            applicationContext.getBean(ReplicationRoutingDataSource.class);
        
        // 3. Update target datasources
        Map<Object, Object> targetDataSources = new HashMap<>();
        targetDataSources.put(DataSourceType.MASTER, newMaster);
        // Keep existing slaves
        targetDataSources.put(DataSourceType.SLAVE, 
            applicationContext.getBean("slave1DataSource"));
        targetDataSources.put(DataSourceType.SLAVE2, 
            applicationContext.getBean("slave2DataSource"));
        
        routingDs.setTargetDataSources(targetDataSources);
        routingDs.afterPropertiesSet();
        
        log.info("Master datasource updated successfully");
    }
    
    // Promote slave to master
    public void promoteSlave(String slaveHost) {
        log.info("Promoting slave to master: {}", slaveHost);
        
        // 1. Update datasource configuration
        updateMasterDataSource(slaveHost);
        
        // 2. Execute post-promotion tasks
        executeSqlOnNewMaster(slaveHost, "STOP SLAVE");
        executeSqlOnNewMaster(slaveHost, "RESET SLAVE ALL");
        executeSqlOnNewMaster(slaveHost, "SET GLOBAL read_only = 0");
        
        // 3. Notify monitoring systems
        notifyFailover(slaveHost);
    }
    
    private void executeSqlOnNewMaster(String host, String sql) {
        // Execute SQL on new master
        try (Connection conn = createConnection(host)) {
            try (Statement stmt = conn.createStatement()) {
                stmt.execute(sql);
            }
        } catch (SQLException e) {
            log.error("Failed to execute SQL on new master: {}", sql, e);
        }
    }
    
    private Connection createConnection(String host) throws SQLException {
        String url = String.format("jdbc:mysql://%s:3306/mydb", host);
        return DriverManager.getConnection(
            url,
            environment.getProperty("spring.datasource.master.username"),
            environment.getProperty("spring.datasource.master.password")
        );
    }
    
    private void notifyFailover(String newMaster) {
        // Send notifications via Slack, PagerDuty, email, etc.
        log.info("ALERT: Failover completed. New master: {}", newMaster);
    }
}
```

### Comprehensive Health Check

```java
@Service
public class ReplicationHealthService {
    
    @Autowired
    private JdbcTemplate masterJdbcTemplate;
    
    @Autowired
    private JdbcTemplate slave1JdbcTemplate;
    
    @Autowired
    private JdbcTemplate slave2JdbcTemplate;
    
    @Autowired
    private MeterRegistry meterRegistry;
    
    @Scheduled(fixedDelay = 10000)  // Every 10 seconds
    public void checkHealth() {
        checkMasterHealth();
        checkSlaveHealth("slave1", slave1JdbcTemplate);
        checkSlaveHealth("slave2", slave2JdbcTemplate);
    }
    
    private void checkMasterHealth() {
        try {
            // Check if master is writable
            String sql = "SELECT @@read_only, @@server_id";
            Map<String, Object> result = masterJdbcTemplate.queryForMap(sql);
            
            int readOnly = (Integer) result.get("@@read_only");
            long serverId = (Long) result.get("@@server_id");
            
            if (readOnly == 1) {
                log.error("Master is in read-only mode! Server ID: {}", serverId);
                meterRegistry.counter("mysql.master.readonly.error").increment();
            }
            
            // Check binary log status
            String binlogSql = "SHOW MASTER STATUS";
            Map<String, Object> binlog = masterJdbcTemplate.queryForMap(binlogSql);
            
            String binlogFile = (String) binlog.get("File");
            long binlogPosition = (Long) binlog.get("Position");
            
            log.debug("Master status: File={}, Position={}", binlogFile, binlogPosition);
            
            meterRegistry.gauge("mysql.master.binlog.position", binlogPosition);
            
        } catch (Exception e) {
            log.error("Failed to check master health", e);
            meterRegistry.counter("mysql.master.health.error").increment();
        }
    }
    
    private void checkSlaveHealth(String slaveName, JdbcTemplate jdbcTemplate) {
        try {
            String sql = "SHOW SLAVE STATUS";
            Map<String, Object> status = jdbcTemplate.queryForMap(sql);
            
            // IO Thread status
            String slaveIoRunning = (String) status.get("Slave_IO_Running");
            String slaveSqlRunning = (String) status.get("Slave_SQL_Running");
            
            if (!"Yes".equals(slaveIoRunning)) {
                log.error("{}: Slave IO thread not running!", slaveName);
                meterRegistry.counter("mysql.slave.io.stopped", "slave", slaveName).increment();
            }
            
            if (!"Yes".equals(slaveSqlRunning)) {
                log.error("{}: Slave SQL thread not running!", slaveName);
                meterRegistry.counter("mysql.slave.sql.stopped", "slave", slaveName).increment();
            }
            
            // Replication lag
            Object secondsBehind = status.get("Seconds_Behind_Master");
            if (secondsBehind != null) {
                long lag = ((Number) secondsBehind).longValue();
                meterRegistry.gauge("mysql.slave.lag.seconds", 
                    Tags.of("slave", slaveName), lag);
                
                if (lag > 30) {
                    log.warn("{}: High replication lag: {} seconds", slaveName, lag);
                }
            }
            
            // Last error
            String lastError = (String) status.get("Last_Error");
            if (lastError != null && !lastError.isEmpty()) {
                log.error("{}: Replication error: {}", slaveName, lastError);
                meterRegistry.counter("mysql.slave.error", "slave", slaveName).increment();
            }
            
            // Master log position (how far behind)
            String masterLogFile = (String) status.get("Master_Log_File");
            long readMasterLogPos = (Long) status.get("Read_Master_Log_Pos");
            long execMasterLogPos = (Long) status.get("Exec_Master_Log_Pos");
            
            log.debug("{}: Master Log File={}, Read Pos={}, Exec Pos={}", 
                slaveName, masterLogFile, readMasterLogPos, execMasterLogPos);
            
        } catch (Exception e) {
            log.error("Failed to check {} health", slaveName, e);
            meterRegistry.counter("mysql.slave.health.error", "slave", slaveName).increment();
        }
    }
    
    public ReplicationHealthReport getHealthReport() {
        ReplicationHealthReport report = new ReplicationHealthReport();
        
        try {
            // Master health
            report.setMasterHealthy(checkMasterReachable());
            
            // Slave health
            Map<String, SlaveHealth> slaveHealth = new HashMap<>();
            slaveHealth.put("slave1", getSlaveHealth("slave1", slave1JdbcTemplate));
            slaveHealth.put("slave2", getSlaveHealth("slave2", slave2JdbcTemplate));
            report.setSlaveHealth(slaveHealth);
            
        } catch (Exception e) {
            log.error("Failed to generate health report", e);
        }
        
        return report;
    }
    
    private boolean checkMasterReachable() {
        try {
            masterJdbcTemplate.queryForObject("SELECT 1", Integer.class);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
    
    private SlaveHealth getSlaveHealth(String name, JdbcTemplate jdbcTemplate) {
        try {
            Map<String, Object> status = jdbcTemplate.queryForMap("SHOW SLAVE STATUS");
            
            SlaveHealth health = new SlaveHealth();
            health.setName(name);
            health.setIoRunning("Yes".equals(status.get("Slave_IO_Running")));
            health.setSqlRunning("Yes".equals(status.get("Slave_SQL_Running")));
            
            Object lag = status.get("Seconds_Behind_Master");
            health.setSecondsBehindMaster(lag != null ? ((Number) lag).longValue() : -1);
            
            health.setLastError((String) status.get("Last_Error"));
            
            return health;
            
        } catch (Exception e) {
            SlaveHealth health = new SlaveHealth();
            health.setName(name);
            health.setReachable(false);
            return health;
        }
    }
}

@Data
class ReplicationHealthReport {
    private boolean masterHealthy;
    private Map<String, SlaveHealth> slaveHealth;
}

@Data
class SlaveHealth {
    private String name;
    private boolean reachable = true;
    private boolean ioRunning;
    private boolean sqlRunning;
    private long secondsBehindMaster;
    private String lastError;
    
    public boolean isHealthy() {
        return reachable && ioRunning && sqlRunning && 
               secondsBehindMaster >= 0 && secondsBehindMaster < 30 &&
               (lastError == null || lastError.isEmpty());
    }
}
```

### Health Check Endpoint

```java
@RestController
@RequestMapping("/admin/health")
public class ReplicationHealthController {
    
    @Autowired
    private ReplicationHealthService healthService;
    
    @GetMapping("/replication")
    public ResponseEntity<ReplicationHealthReport> getReplicationHealth() {
        ReplicationHealthReport report = healthService.getHealthReport();
        
        // Return 503 if any critical issue
        if (!report.isMasterHealthy()) {
            return ResponseEntity.status(HttpStatus.SERVICE_UNAVAILABLE).body(report);
        }
        
        // Check if all slaves are healthy
        boolean allSlavesHealthy = report.getSlaveHealth().values().stream()
            .allMatch(SlaveHealth::isHealthy);
        
        if (!allSlavesHealthy) {
            return ResponseEntity.status(HttpStatus.SERVICE_UNAVAILABLE).body(report);
        }
        
        return ResponseEntity.ok(report);
    }
    
    @GetMapping("/replication/lag")
    public ResponseEntity<Map<String, Long>> getReplicationLag() {
        Map<String, Long> lag = new HashMap<>();
        
        ReplicationHealthReport report = healthService.getHealthReport();
        report.getSlaveHealth().forEach((name, health) -> 
            lag.put(name, health.getSecondsBehindMaster())
        );
        
        return ResponseEntity.ok(lag);
    }
}
```

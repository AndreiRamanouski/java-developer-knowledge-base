# Database Indexing Strategies in PostgreSQL

## Deep Dive Interview Guide

This guide covers PostgreSQL indexing strategies with Spring Boot, including index types, creation strategies, optimization techniques, maintenance, and monitoring.

**Interview Tip**: Indexing questions test understanding of query optimization, database internals, performance tuning, and production maintenance.

---

## Q5: Database indexing strategies

**Short Answer**: Create indexes based on query patterns (WHERE, JOIN, ORDER BY). B-tree (default, 95% of cases), Hash (equality only, rare), GiST (geometric, full-text), GIN (arrays, JSONB, full-text). Composite index column order matters: equality first, then range, then sort. Use partial indexes for filtered queries. Maintain with VACUUM, REINDEX when bloated. Analyze with EXPLAIN ANALYZE. Monitor bloat with pg_stat_user_indexes.

## 1. When to Create Indexes (Query Pattern Analysis)

### Understanding Index Need

```sql
-- Find slow queries (candidates for indexing)
SELECT 
    query,
    calls,
    total_exec_time,
    mean_exec_time,
    max_exec_time,
    rows,
    100.0 * shared_blks_hit / NULLIF(shared_blks_hit + shared_blks_read, 0) AS cache_hit_ratio
FROM pg_stat_statements
WHERE mean_exec_time > 10  -- Queries taking > 10ms average
  AND calls > 100          -- Executed frequently
ORDER BY mean_exec_time * calls DESC  -- Total time impact
LIMIT 20;

-- Result example:
-- query: SELECT * FROM orders WHERE customer_id = $1
-- calls: 10000
-- mean_exec_time: 45ms
-- total_impact: 450,000ms (7.5 minutes!)
-- cache_hit_ratio: 75%
-- Action: Create index on customer_id
```

### Query Pattern Analysis

```java
// Analyze application query patterns

@Component
public class QueryPatternAnalyzer {
    
    @Autowired
    private JdbcTemplate jdbcTemplate;
    
    public List<QueryPattern> analyzePatterns() {
        String sql = """
            SELECT 
                query,
                calls,
                mean_exec_time,
                CASE 
                    WHEN query LIKE '%WHERE%=%' THEN 'EQUALITY'
                    WHEN query LIKE '%WHERE%LIKE%' THEN 'PATTERN_MATCH'
                    WHEN query LIKE '%WHERE%IN%' THEN 'IN_CLAUSE'
                    WHEN query LIKE '%WHERE%>%' OR query LIKE '%WHERE%<%' THEN 'RANGE'
                    WHEN query LIKE '%ORDER BY%' THEN 'SORT'
                    WHEN query LIKE '%JOIN%' THEN 'JOIN'
                    ELSE 'OTHER'
                END as pattern_type
            FROM pg_stat_statements
            WHERE calls > 50
            ORDER BY mean_exec_time * calls DESC
            LIMIT 50
            """;
        
        return jdbcTemplate.query(sql, (rs, rowNum) -> {
            QueryPattern pattern = new QueryPattern();
            pattern.setQuery(rs.getString("query"));
            pattern.setCalls(rs.getLong("calls"));
            pattern.setMeanTime(rs.getDouble("mean_exec_time"));
            pattern.setPatternType(rs.getString("pattern_type"));
            return pattern;
        });
    }
    
    public List<IndexRecommendation> recommendIndexes(List<QueryPattern> patterns) {
        List<IndexRecommendation> recommendations = new ArrayList<>();
        
        for (QueryPattern pattern : patterns) {
            // Parse query to extract table and columns
            String query = pattern.getQuery().toLowerCase();
            
            if (pattern.getPatternType().equals("EQUALITY")) {
                // WHERE column = value
                // Recommendation: B-tree index on column
                recommendations.add(new IndexRecommendation(
                    "B-tree",
                    extractTable(query),
                    extractWhereColumns(query),
                    "High impact equality search"
                ));
            } else if (pattern.getPatternType().equals("RANGE")) {
                // WHERE column > value AND column < value
                // Recommendation: B-tree index (supports range)
                recommendations.add(new IndexRecommendation(
                    "B-tree",
                    extractTable(query),
                    extractWhereColumns(query),
                    "Range query optimization"
                ));
            } else if (pattern.getPatternType().equals("PATTERN_MATCH")) {
                // WHERE column LIKE 'prefix%'
                // Recommendation: B-tree with text_pattern_ops
                recommendations.add(new IndexRecommendation(
                    "B-tree with text_pattern_ops",
                    extractTable(query),
                    extractLikeColumns(query),
                    "Prefix pattern matching"
                ));
            } else if (pattern.getPatternType().equals("JOIN")) {
                // JOIN ON foreign_key = primary_key
                // Recommendation: Index on foreign key
                recommendations.add(new IndexRecommendation(
                    "B-tree",
                    extractJoinTable(query),
                    extractJoinColumns(query),
                    "Join optimization"
                ));
            }
        }
        
        return deduplicateAndPrioritize(recommendations);
    }
}

@Data
class QueryPattern {
    private String query;
    private Long calls;
    private Double meanTime;
    private String patternType;
    private Double totalImpact;
}

@Data
@AllArgsConstructor
class IndexRecommendation {
    private String indexType;
    private String tableName;
    private List<String> columns;
    private String reason;
    private Double expectedImpact;
}
```

### Index Creation Rules

```sql
-- Rule 1: Index foreign keys (for JOINs)
-- Every foreign key should have an index

-- Find foreign keys without indexes
SELECT
    tc.table_name,
    kcu.column_name,
    ccu.table_name AS foreign_table_name
FROM information_schema.table_constraints AS tc
JOIN information_schema.key_column_usage AS kcu
    ON tc.constraint_name = kcu.constraint_name
JOIN information_schema.constraint_column_usage AS ccu
    ON ccu.constraint_name = tc.constraint_name
WHERE tc.constraint_type = 'FOREIGN KEY'
  AND NOT EXISTS (
      SELECT 1
      FROM pg_indexes
      WHERE tablename = tc.table_name
        AND indexdef LIKE '%' || kcu.column_name || '%'
  );

-- Create missing indexes
CREATE INDEX idx_orders_customer_id ON orders(customer_id);
CREATE INDEX idx_order_items_order_id ON order_items(order_id);
CREATE INDEX idx_order_items_product_id ON order_items(product_id);

-- Rule 2: Index WHERE clause columns
-- Columns frequently used in WHERE conditions

-- Example queries:
SELECT * FROM orders WHERE status = 'PENDING';
SELECT * FROM users WHERE email = 'user@example.com';
SELECT * FROM products WHERE price > 100 AND price < 500;

-- Create indexes:
CREATE INDEX idx_orders_status ON orders(status);
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_products_price ON products(price);

-- Rule 3: Index JOIN columns
-- Both sides of JOIN conditions

SELECT o.*, c.name
FROM orders o
JOIN customers c ON o.customer_id = c.id;  -- Index both columns

CREATE INDEX idx_orders_customer_id ON orders(customer_id);  -- Usually foreign key
-- customers.id is already indexed (primary key)

-- Rule 4: Index ORDER BY columns
-- Columns used for sorting

SELECT * FROM products ORDER BY created_at DESC LIMIT 20;

CREATE INDEX idx_products_created_at ON products(created_at DESC);

-- Rule 5: Index GROUP BY columns
-- Columns used for aggregation

SELECT customer_id, COUNT(*)
FROM orders
GROUP BY customer_id;

CREATE INDEX idx_orders_customer_id ON orders(customer_id);
```

### When NOT to Create Indexes

```sql
-- ❌ DON'T index:

-- 1. Small tables (< 1000 rows)
-- Index overhead > scan time
SELECT * FROM countries;  -- 195 rows, no index needed on name

-- 2. High write:read ratio tables
-- More index maintenance cost than query benefit
-- Example: audit logs (1000 writes/sec, 1 read/day)

-- 3. Low cardinality columns (few distinct values)
-- Example: gender (M/F/O), boolean flags
CREATE INDEX idx_users_gender ON users(gender);  -- BAD! Only 3 values
-- Exception: Partial indexes (see later section)

-- 4. Columns that are frequently updated
-- Index must be updated on every change
CREATE INDEX idx_products_view_count ON products(view_count);  -- BAD! Updates every view

-- 5. Wide columns
-- Large indexes consume memory and slow updates
CREATE INDEX idx_posts_content ON posts(content);  -- BAD! Content is large text

-- 6. Columns never used in queries
-- Wasted space and maintenance overhead
CREATE INDEX idx_users_last_login ON users(last_login);  -- Only if queried!

-- Check index usage
SELECT
    schemaname,
    tablename,
    indexname,
    idx_scan,  -- Number of times index was used
    idx_tup_read,  -- Tuples read from index
    idx_tup_fetch  -- Tuples fetched from table
FROM pg_stat_user_indexes
WHERE idx_scan = 0  -- Never used!
  AND schemaname = 'public'
ORDER BY pg_relation_size(indexrelid) DESC;  -- Largest unused indexes
```

## 2. B-tree vs Hash vs GiST vs GIN Indexes

### B-tree Index (Default, 95% of Cases)

```sql
-- B-tree: Balanced tree structure, supports:
-- - Equality (=)
-- - Inequality (<, <=, >, >=)
-- - BETWEEN
-- - IS NULL / IS NOT NULL
-- - LIKE 'prefix%' (with C locale or text_pattern_ops)
-- - ORDER BY

-- Create B-tree index (default)
CREATE INDEX idx_orders_created_at ON orders(created_at);
CREATE INDEX idx_users_email ON users(email);

-- B-tree with DESC for descending sorts
CREATE INDEX idx_products_price_desc ON products(price DESC);

-- Query using B-tree index
SELECT * FROM orders WHERE created_at >= '2024-01-01';  -- Uses index
SELECT * FROM orders WHERE created_at BETWEEN '2024-01-01' AND '2024-12-31';  -- Uses index
SELECT * FROM orders ORDER BY created_at DESC LIMIT 10;  -- Uses index

-- Check index type
SELECT
    tablename,
    indexname,
    indexdef
FROM pg_indexes
WHERE tablename = 'orders';

-- Result:
-- indexname: idx_orders_created_at
-- indexdef: CREATE INDEX idx_orders_created_at ON orders USING btree (created_at)
```

**B-tree Performance**

```java
@SpringBootTest
public class BtreePerformanceTest {
    
    @Test
    public void testBtreePerformance() {
        // Setup: 1 million orders
        setupOrders(1_000_000);
        
        // Test 1: Sequential scan (no index)
        dropIndex("idx_orders_created_at");
        long start = System.currentTimeMillis();
        jdbcTemplate.queryForList(
            "SELECT * FROM orders WHERE created_at > '2024-01-01'"
        );
        long seqScan = System.currentTimeMillis() - start;
        
        // Test 2: Index scan (with B-tree)
        createIndex("CREATE INDEX idx_orders_created_at ON orders(created_at)");
        start = System.currentTimeMillis();
        jdbcTemplate.queryForList(
            "SELECT * FROM orders WHERE created_at > '2024-01-01'"
        );
        long indexScan = System.currentTimeMillis() - start;
        
        System.out.println("Sequential scan: " + seqScan + "ms");  // 850ms
        System.out.println("Index scan: " + indexScan + "ms");     // 35ms
        System.out.println("Speedup: " + (seqScan / indexScan) + "x");  // 24x
        
        // B-tree index: 24x faster for range queries
    }
}
```

### Hash Index (Equality Only, Rarely Used)

```sql
-- Hash index: Hash table, supports ONLY equality (=)
-- Faster than B-tree for equality, but:
-- - Not WAL-logged (not crash-safe) in PostgreSQL < 10
-- - Cannot be used for <, >, BETWEEN, ORDER BY
-- - Generally avoid unless specific use case

-- Create hash index
CREATE INDEX idx_users_email_hash ON users USING hash(email);

-- Queries that can use hash index
SELECT * FROM users WHERE email = 'user@example.com';  -- Uses hash index

-- Queries that CANNOT use hash index
SELECT * FROM users WHERE email LIKE 'user%';  -- Cannot use hash
SELECT * FROM users WHERE email > 'a@example.com';  -- Cannot use hash
SELECT * FROM users ORDER BY email;  -- Cannot use hash

-- Performance comparison
-- B-tree equality:  O(log n) = 20 comparisons for 1M rows
-- Hash equality:    O(1) = 1 hash lookup
-- Difference: ~5-10% faster (usually not worth it)

-- Recommendation: Just use B-tree (more versatile)
```

### GiST Index (Geometric, Full-Text Search)

```sql
-- GiST: Generalized Search Tree
-- Supports:
-- - Geometric types (point, box, circle, polygon)
-- - Range types (tsrange, int4range)
-- - Full-text search (tsvector)
-- - Exclusion constraints

-- Example 1: Geometric data
CREATE TABLE locations (
    id SERIAL PRIMARY KEY,
    name TEXT,
    coordinates POINT
);

CREATE INDEX idx_locations_coordinates ON locations USING gist(coordinates);

-- Query: Find locations within radius
SELECT * FROM locations
WHERE coordinates <-> point(40.7128, -74.0060) < 10;  -- Within 10 units

-- Example 2: Range types
CREATE TABLE bookings (
    id SERIAL PRIMARY KEY,
    room_id INT,
    time_range TSRANGE
);

CREATE INDEX idx_bookings_time_range ON bookings USING gist(time_range);

-- Query: Find overlapping bookings
SELECT * FROM bookings
WHERE time_range && tsrange('2024-01-01', '2024-01-02');  -- Overlaps

-- Exclude constraint (prevent overlapping bookings)
ALTER TABLE bookings
ADD CONSTRAINT no_overlap
EXCLUDE USING gist (room_id WITH =, time_range WITH &&);

-- Example 3: Full-text search (alternative to GIN)
CREATE INDEX idx_posts_content_gist ON posts USING gist(to_tsvector('english', content));

SELECT * FROM posts
WHERE to_tsvector('english', content) @@ to_tsquery('postgresql & indexing');
```

### GIN Index (Arrays, JSONB, Full-Text)

```sql
-- GIN: Generalized Inverted Index
-- Optimized for:
-- - Arrays
-- - JSONB
-- - Full-text search (tsvector)
-- - Multiple values per row

-- Example 1: Array columns
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    name TEXT,
    tags TEXT[]
);

CREATE INDEX idx_products_tags ON products USING gin(tags);

-- Query: Find products with specific tag
SELECT * FROM products WHERE tags @> ARRAY['electronics'];
SELECT * FROM products WHERE tags && ARRAY['electronics', 'sale'];  -- Any of

-- Example 2: JSONB columns
CREATE TABLE events (
    id SERIAL PRIMARY KEY,
    data JSONB
);

CREATE INDEX idx_events_data ON events USING gin(data);

-- Query: Search JSONB
SELECT * FROM events WHERE data @> '{"user_id": 123}';
SELECT * FROM events WHERE data ? 'error';  -- Key exists
SELECT * FROM events WHERE data ?| array['error', 'warning'];  -- Any key

-- GIN with specific JSONB path
CREATE INDEX idx_events_user_id ON events USING gin((data -> 'user_id'));

-- Example 3: Full-text search (most common for GIN)
CREATE TABLE articles (
    id SERIAL PRIMARY KEY,
    title TEXT,
    content TEXT,
    search_vector TSVECTOR
);

-- Create GIN index on tsvector
CREATE INDEX idx_articles_search ON articles USING gin(search_vector);

-- Populate search vector (trigger or generated column)
UPDATE articles SET search_vector = 
    to_tsvector('english', coalesce(title, '') || ' ' || coalesce(content, ''));

-- Query: Full-text search
SELECT * FROM articles
WHERE search_vector @@ to_tsquery('english', 'postgresql & performance');

-- With ranking
SELECT *, ts_rank(search_vector, query) AS rank
FROM articles, to_tsquery('english', 'postgresql & performance') AS query
WHERE search_vector @@ query
ORDER BY rank DESC;
```

### Index Type Comparison

```sql
-- Performance comparison: 1M rows

-- B-tree:
-- Equality (=):        O(log n) = ~20 comparisons
-- Range (<, >):        O(log n) = ~20 comparisons
-- Sort (ORDER BY):     Uses index directly
-- Space:               Moderate (100-200% of data size)
-- Update cost:         Low (single node update)
-- Use case:            95% of indexes

-- Hash:
-- Equality (=):        O(1) = 1 hash lookup
-- Range (<, >):        Cannot use
-- Sort (ORDER BY):     Cannot use
-- Space:               Small (50-100% of data size)
-- Update cost:         Low (single hash bucket)
-- Use case:            Rare (only when absolutely need 5% speed gain)

-- GiST:
-- Equality (=):        O(log n)
-- Range (&&, <->):     O(log n)
-- Containment (@>):    O(log n)
-- Space:               Large (200-500% of data size)
-- Update cost:         High (tree rebalancing)
-- Use case:            Geometric, ranges, full-text (when update-heavy)

-- GIN:
-- Containment (@>):    O(1) for each element
-- Array intersection:  O(n) where n = array size
-- Full-text:           O(m) where m = matching documents
-- Space:               Very large (300-1000% of data size)
-- Update cost:         Very high (update all entries)
-- Use case:            Arrays, JSONB, full-text (when read-heavy)
```

**Decision Tree**

```java
public IndexType chooseIndexType(Column column) {
    // B-tree (default)
    if (column.isNumeric() || column.isDate() || column.isText()) {
        return IndexType.BTREE;
    }
    
    // Hash (rare)
    if (column.isHighCardinality() && 
        column.isEquality Only() && 
        column.isNeverSorted()) {
        return IndexType.HASH;  // Consider, but usually B-tree is fine
    }
    
    // GiST
    if (column.isGeometric() || column.isRange()) {
        return IndexType.GIST;
    }
    
    if (column.isFullText() && column.isUpdateHeavy()) {
        return IndexType.GIST;  // Better for frequent updates
    }
    
    // GIN
    if (column.isArray() || column.isJSONB()) {
        return IndexType.GIN;
    }
    
    if (column.isFullText() && column.isReadHeavy()) {
        return IndexType.GIN;  // Better for frequent searches
    }
    
    // Default
    return IndexType.BTREE;
}
```

## 3. Composite Indexes and Column Order

### Column Order Matters

```sql
-- Wrong order example
CREATE INDEX idx_orders_date_customer ON orders(created_at, customer_id);

-- Query 1: Uses index (leftmost prefix)
SELECT * FROM orders WHERE created_at > '2024-01-01' AND customer_id = 123;

-- Query 2: Uses index (leftmost column)
SELECT * FROM orders WHERE created_at > '2024-01-01';

-- Query 3: CANNOT use index (missing leftmost column)
SELECT * FROM orders WHERE customer_id = 123;  -- Sequential scan!

-- Correct order: Most selective / equality first
CREATE INDEX idx_orders_customer_date ON orders(customer_id, created_at);

-- Now Query 3 can use index!
SELECT * FROM orders WHERE customer_id = 123;  -- Index scan

-- And Query 1 still works
SELECT * FROM orders WHERE customer_id = 123 AND created_at > '2024-01-01';
```

### Column Order Rules

```sql
-- Rule 1: Equality columns before range columns

-- BAD: Range column first
CREATE INDEX idx_orders_bad ON orders(created_at, status);

SELECT * FROM orders 
WHERE created_at BETWEEN '2024-01-01' AND '2024-12-31'
  AND status = 'COMPLETED';
-- Can use index, but not efficiently (scans entire date range, then filters status)

-- GOOD: Equality first, then range
CREATE INDEX idx_orders_good ON orders(status, created_at);

SELECT * FROM orders 
WHERE status = 'COMPLETED'
  AND created_at BETWEEN '2024-01-01' AND '2024-12-31';
-- Efficient: Jumps to status='COMPLETED', then scans date range within that subset

-- Rule 2: Most selective column first (for equality)

-- Analyze selectivity
SELECT 
    COUNT(DISTINCT customer_id) as distinct_customers,  -- 10,000
    COUNT(DISTINCT status) as distinct_statuses         -- 5
FROM orders;

-- customer_id is 2,000x more selective (10,000 / 5)

-- GOOD: Most selective first
CREATE INDEX idx_orders_customer_status ON orders(customer_id, status);

-- Query: Both columns in WHERE
SELECT * FROM orders WHERE customer_id = 123 AND status = 'PENDING';
-- Efficient: Jumps to customer_id=123 (narrows to 10 rows), then filters status

-- BAD: Least selective first
CREATE INDEX idx_orders_status_customer ON orders(status, customer_id);
-- Same query: Jumps to status='PENDING' (narrows to 200,000 rows), then filters customer
-- Much more work!

-- Rule 3: ORDER BY columns last

-- Query: Filter + sort
SELECT * FROM products 
WHERE category_id = 5 
ORDER BY created_at DESC 
LIMIT 20;

-- GOOD: Filter columns first, then sort column
CREATE INDEX idx_products_category_created ON products(category_id, created_at DESC);

-- Rule 4: Include all columns needed for covering index (INCLUDE)

-- Query: Need specific columns
SELECT id, name, price 
FROM products 
WHERE category_id = 5 
ORDER BY created_at DESC;

-- Option 1: Regular composite index (all columns in index)
CREATE INDEX idx_products_composite ON products(category_id, created_at DESC, id, name, price);

-- Option 2: INCLUDE clause (index + included columns)
CREATE INDEX idx_products_covering ON products(category_id, created_at DESC) 
INCLUDE (id, name, price);

-- INCLUDE advantage:
-- - Included columns don't participate in index structure
-- - Smaller index (faster scans)
-- - Still covering (index-only scan)
```

### Composite Index Examples

```sql
-- Example 1: E-commerce order search

-- Common query
SELECT * FROM orders
WHERE customer_id = ?
  AND status = ?
  AND created_at > ?
ORDER BY created_at DESC;

-- Analyze query pattern:
-- 1. customer_id = ? (equality, high selectivity - 10K customers)
-- 2. status = ? (equality, low selectivity - 5 statuses)
-- 3. created_at > ? (range)
-- 4. ORDER BY created_at DESC (sort)

-- Optimal index (equality, selectivity, range, sort):
CREATE INDEX idx_orders_search ON orders(
    customer_id,    -- Equality, most selective (10K values)
    status,         -- Equality, less selective (5 values)
    created_at DESC -- Range + sort
);

-- Example 2: Multi-tenant application

-- Query
SELECT * FROM documents
WHERE tenant_id = ?
  AND user_id = ?
  AND deleted = false
ORDER BY updated_at DESC
LIMIT 20;

-- Analyze:
-- 1. tenant_id (equality, medium selectivity - 100 tenants)
-- 2. user_id (equality, high selectivity - 100K users)
-- 3. deleted (equality, very low selectivity - boolean)
-- 4. ORDER BY updated_at DESC (sort)

-- Optimal index:
CREATE INDEX idx_documents_search ON documents(
    tenant_id,      -- Partition by tenant (isolation)
    user_id,        -- Most selective equality
    deleted,        -- Boolean (but in WHERE clause)
    updated_at DESC -- Sort
) WHERE deleted = false;  -- Partial index (see next section)

-- Example 3: Analytics dashboard

-- Query: Aggregation with filter
SELECT 
    product_id,
    COUNT(*),
    SUM(amount)
FROM sales
WHERE sale_date >= '2024-01-01'
  AND region = 'US'
GROUP BY product_id;

-- Index for GROUP BY + WHERE:
CREATE INDEX idx_sales_analytics ON sales(
    region,         -- Filter (equality)
    sale_date,      -- Filter (range)
    product_id      -- GROUP BY
);
```

## 4. Partial Indexes (WHERE Clause)

### Basic Partial Indexes

```sql
-- Partial index: Index only subset of rows

-- Example 1: Active users only
CREATE INDEX idx_users_email_active ON users(email) 
WHERE active = true;

-- Query that uses partial index
SELECT * FROM users WHERE email = 'user@example.com' AND active = true;
-- Uses idx_users_email_active (smaller, faster)

-- Query that CANNOT use partial index
SELECT * FROM users WHERE email = 'user@example.com';
-- Cannot use idx_users_email_active (missing WHERE active = true)
-- Must use full index or sequential scan

-- Example 2: Pending orders only
CREATE INDEX idx_orders_customer_pending ON orders(customer_id) 
WHERE status = 'PENDING';

-- Benefits:
-- 1. Smaller index (only 10% of orders are pending)
-- 2. Faster queries (less data to scan)
-- 3. Less maintenance (updates to completed orders don't touch index)

-- Regular index size: 500MB (all orders)
-- Partial index size: 50MB (pending orders only)
-- 10x smaller!

-- Example 3: Recent data only
CREATE INDEX idx_events_user_recent ON events(user_id, created_at) 
WHERE created_at > NOW() - INTERVAL '30 days';

-- Benefits for time-series data:
-- - Index only hot data
-- - Old data uses sequential scan (acceptable, rarely queried)
-- - Index stays small and fast
```

### Partial Index Use Cases

```sql
-- Use Case 1: Soft deletes

-- Table with soft delete
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    name TEXT,
    price DECIMAL,
    deleted_at TIMESTAMP  -- NULL = not deleted
);

-- Regular index (includes deleted products)
CREATE INDEX idx_products_name ON products(name);
-- Size: 100MB (1M products)

-- Partial index (only active products)
CREATE INDEX idx_products_name_active ON products(name) 
WHERE deleted_at IS NULL;
-- Size: 90MB (900K active products)

-- Query (with WHERE deleted_at IS NULL)
SELECT * FROM products WHERE name LIKE 'Laptop%' AND deleted_at IS NULL;
-- Uses partial index (10% smaller, faster)

-- Use Case 2: Multi-tenant with selective indexing

-- Large tenant needs fast queries
CREATE INDEX idx_documents_large_tenant ON documents(user_id, created_at) 
WHERE tenant_id = 12345;  -- Large tenant with 80% of data

-- Small tenants share a general index
CREATE INDEX idx_documents_other_tenants ON documents(tenant_id, user_id) 
WHERE tenant_id != 12345;

-- Use Case 3: Status-based indexing

-- Most rows are in 'COMPLETED' status (90%)
-- Only 'PENDING' and 'PROCESSING' need fast lookup

CREATE INDEX idx_orders_status_active ON orders(customer_id, created_at) 
WHERE status IN ('PENDING', 'PROCESSING');

-- Index size: 10x smaller
-- Query speed: Same as full index for pending/processing
-- Update speed: 9x faster (90% of updates don't touch index)

-- Use Case 4: Boolean columns

-- Don't index false values (90% of rows)
CREATE INDEX idx_users_premium_true ON users(id) 
WHERE is_premium = true;

-- Query for premium users
SELECT * FROM users WHERE is_premium = true AND last_login > NOW() - INTERVAL '7 days';

-- Regular index on boolean: wasteful (indexes false values too)
-- Partial index: efficient (only true values)
```

## 5. Index Maintenance (REINDEX, VACUUM)

### Understanding Index Bloat

```sql
-- Index bloat occurs when:
-- 1. Many updates/deletes create dead tuples
-- 2. VACUUM doesn't reclaim space efficiently
-- 3. Index pages become sparse

-- Check index bloat
SELECT
    schemaname,
    tablename,
    indexname,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size,
    idx_scan,
    idx_tup_read,
    idx_tup_fetch,
    pg_size_pretty(
        pg_relation_size(indexrelid) - 
        (pg_relation_size(indexrelid) * 
         COALESCE(
             (SELECT (100 - pg_stat_get_live_tuples(indexrelid)::float / 
                     NULLIF(pg_stat_get_tuples_inserted(indexrelid) + 
                            pg_stat_get_tuples_updated(indexrelid) + 
                            pg_stat_get_tuples_deleted(indexrelid), 0) * 100
             ), 0
         ) / 100)
    ) AS bloat
FROM pg_stat_user_indexes
WHERE schemaname = 'public'
ORDER BY pg_relation_size(indexrelid) DESC;

-- More accurate bloat check (pgstattuple extension)
CREATE EXTENSION IF NOT EXISTS pgstattuple;

SELECT
    indexrelname,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size,
    (pgstatindex(indexrelid)).avg_leaf_density AS leaf_density,
    (pgstatindex(indexrelid)).leaf_fragmentation AS fragmentation
FROM pg_stat_user_indexes
WHERE schemaname = 'public';

-- Healthy index: leaf_density > 90%, fragmentation < 20%
-- Bloated index: leaf_density < 70%, fragmentation > 50%
```

### VACUUM

```sql
-- VACUUM: Reclaims space from dead tuples

-- 1. Regular VACUUM (non-blocking)
VACUUM orders;
-- - Marks dead tuples as reusable
-- - Updates statistics
-- - Doesn't shrink files (space reused for new data)
-- - Can run concurrently with queries

-- 2. VACUUM ANALYZE (recommended)
VACUUM ANALYZE orders;
-- - VACUUM + updates query planner statistics
-- - Should be run regularly (daily/hourly for busy tables)

-- 3. VACUUM FULL (blocking, dangerous)
VACUUM FULL orders;
-- - Reclaims space and shrinks files
-- - LOCKS table (blocks all queries!)
-- - Rewrites entire table
-- - Use only during maintenance windows

-- 4. VACUUM VERBOSE (for debugging)
VACUUM VERBOSE orders;
-- Output:
-- INFO: vacuuming "public.orders"
-- INFO: "orders": found 50000 removable, 950000 nonremovable row versions in 100000 pages
-- INFO: "orders": removed 50000 dead row versions in 5000 pages

-- Autovacuum configuration
SELECT
    relname,
    last_vacuum,
    last_autovacuum,
    n_dead_tup,
    n_live_tup,
    n_dead_tup::float / NULLIF(n_live_tup, 0) AS dead_ratio
FROM pg_stat_user_tables
WHERE schemaname = 'public'
ORDER BY n_dead_tup DESC;

-- Tune autovacuum for specific table
ALTER TABLE orders SET (
    autovacuum_vacuum_threshold = 100,      -- Start vacuum after 100 dead tuples
    autovacuum_vacuum_scale_factor = 0.05,  -- Or 5% of table size (default 20%)
    autovacuum_vacuum_cost_delay = 10,      -- Slow down vacuum (less aggressive)
    autovacuum_analyze_threshold = 50,
    autovacuum_analyze_scale_factor = 0.05
);
```

### REINDEX

```sql
-- REINDEX: Rebuilds index from scratch

-- When to REINDEX:
-- 1. Index is bloated (>50% fragmentation)
-- 2. After bulk updates/deletes
-- 3. Corruption suspected
-- 4. To change index parameters

-- 1. REINDEX single index
REINDEX INDEX idx_orders_customer_id;
-- - Rebuilds one index
-- - Locks table (blocks writes, allows reads with CONCURRENTLY)

-- 2. REINDEX table (all indexes)
REINDEX TABLE orders;
-- - Rebuilds all indexes on table
-- - Locks table

-- 3. REINDEX CONCURRENTLY (PostgreSQL 12+)
REINDEX INDEX CONCURRENTLY idx_orders_customer_id;
-- - Rebuilds index without locking
-- - Takes longer
-- - Requires 2x space temporarily
-- - Safe for production

-- 4. REINDEX database
REINDEX DATABASE mydb;
-- - Rebuilds all indexes in database
-- - Very long operation
-- - Use during maintenance window

-- Monitor REINDEX progress
SELECT
    pid,
    now() - query_start AS duration,
    query
FROM pg_stat_activity
WHERE query LIKE 'REINDEX%';
```

### Common Scan Types

```sql
-- 1. Sequential Scan (table scan)
-- Reads entire table from disk
-- Cost: O(n) where n = total rows
-- When: No index, or table is small (< 1000 rows)

EXPLAIN ANALYZE
SELECT * FROM small_table;
-- Seq Scan on small_table  (cost=0.00..10.50 rows=500 width=100)

-- 2. Index Scan
-- Uses index to find rows, then fetches from table
-- Cost: O(log n + k) where k = matching rows
-- When: Selective query (<5% of table)

EXPLAIN ANALYZE
SELECT * FROM orders WHERE customer_id = 123;
-- Index Scan using idx_orders_customer_id on orders

-- 3. Index Only Scan
-- Uses index without accessing table (covering index)
-- Cost: O(log n + k), but faster (no table access)
-- When: All columns in SELECT are in index

EXPLAIN ANALYZE
SELECT id, customer_id FROM orders WHERE customer_id = 123;
-- Index Only Scan using idx_orders_customer_id on orders

-- 4. Bitmap Index Scan + Bitmap Heap Scan
-- Creates bitmap of matching rows, then fetches in physical order
-- Cost: O(log n + k), optimized for disk I/O
-- When: Moderate selectivity (5-25% of table)

EXPLAIN ANALYZE
SELECT * FROM orders WHERE status = 'PENDING';
-- Bitmap Heap Scan on orders
--   -> Bitmap Index Scan on idx_orders_status

-- 5. Hash Join
-- Builds hash table of smaller table, probes with larger table
-- Cost: O(n + m)
-- When: Joining tables of different sizes

EXPLAIN ANALYZE
SELECT * FROM orders o JOIN customers c ON o.customer_id = c.id;
-- Hash Join
--   Hash Cond: (o.customer_id = c.id)
--   -> Seq Scan on orders o
--   -> Hash
--        -> Seq Scan on customers c

-- 6. Nested Loop
-- For each row in outer table, scan inner table
-- Cost: O(n * m)
-- When: Small tables, or highly selective joins

EXPLAIN ANALYZE
SELECT * FROM orders o JOIN customers c ON o.customer_id = c.id
WHERE o.id = 123;
-- Nested Loop
--   -> Index Scan on orders o (1 row)
--   -> Index Scan on customers c (1 row)

-- 7. Merge Join
-- Sorts both tables, then merges
-- Cost: O(n log n + m log m)
-- When: Both tables sorted (or have suitable indexes)

EXPLAIN ANALYZE
SELECT * FROM orders o JOIN customers c ON o.customer_id = c.id
ORDER BY o.customer_id;
-- Merge Join
--   Merge Cond: (c.id = o.customer_id)
--   -> Index Scan on customers c
--   -> Index Scan on orders o
```

### Identifying Problems

```sql
-- Problem 1: Sequential Scan (should use index)

EXPLAIN ANALYZE
SELECT * FROM orders WHERE status = 'PENDING';
-- Seq Scan on orders  (cost=0.00..25000.00 rows=100000 width=100) (actual time=250.123..500.234 rows=100000 loops=1)

-- Solution: Create index
CREATE INDEX idx_orders_status ON orders(status);

-- Problem 2: Index not used (wrong data type)

EXPLAIN ANALYZE
SELECT * FROM orders WHERE id = '123';  -- String instead of int!
-- Seq Scan on orders  (cost=0.00..25000.00 rows=1 width=100)
--   Filter: ((id)::text = '123'::text)

-- Solution: Use correct type
SELECT * FROM orders WHERE id = 123;  -- Now uses index

-- Problem 3: Function on indexed column

EXPLAIN ANALYZE
SELECT * FROM users WHERE LOWER(email) = 'user@example.com';
-- Seq Scan on users  (cost=0.00..1500.00 rows=1 width=50)
--   Filter: (lower(email) = 'user@example.com'::text)

-- Solution: Functional index
CREATE INDEX idx_users_email_lower ON users(LOWER(email));

-- Problem 4: OR condition (can't use single index efficiently)

EXPLAIN ANALYZE
SELECT * FROM orders WHERE customer_id = 123 OR status = 'PENDING';
-- Seq Scan on orders  (cost=0.00..25000.00 rows=5000 width=100)

-- Solution: Union or separate queries
SELECT * FROM orders WHERE customer_id = 123
UNION ALL
SELECT * FROM orders WHERE status = 'PENDING' AND customer_id != 123;

-- Problem 5: Underestimated cardinality (bad statistics)

EXPLAIN ANALYZE
SELECT * FROM orders WHERE status = 'RARE_STATUS';
-- Seq Scan on orders  (cost=0.00..25000.00 rows=1000 width=100) (actual rows=10 loops=1)
--   -- Estimated 1000 rows, actual 10 rows (100x off!)

-- Solution: Update statistics
ANALYZE orders;

-- Or increase statistics target
ALTER TABLE orders ALTER COLUMN status SET STATISTICS 1000;
ANALYZE orders;
```

## Interview Follow-Up Questions - Detailed Answers

### Q1: "How would you identify which columns need indexes in a new application?"

**Answer**: Use 4-step approach: 1) Analyze query patterns (WHERE, JOIN, ORDER BY columns), 2) Enable pg_stat_statements to track slow queries, 3) Review application code for common queries, 4) Start with foreign keys and high-cardinality columns in WHERE clauses. Monitor usage after creation and drop unused indexes.

**Step-by-Step Process**

```java
// Step 1: Query Pattern Analysis from Code

@Component
public class IndexIdentificationService {
    
    @Autowired
    private ApplicationContext applicationContext;
    
    public Map<String, List<String>> analyzeRepositoryQueries() {
        Map<String, List<String>> recommendations = new HashMap<>();
        
        // Scan all repositories
        Map<String, Object> repositories = applicationContext
            .getBeansWithAnnotation(Repository.class);
        
        for (Object repo : repositories.values()) {
            Class<?> repoClass = repo.getClass();
            
            // Analyze @Query annotations
            for (Method method : repoClass.getDeclaredMethods()) {
                Query queryAnnotation = method.getAnnotation(Query.class);
                if (queryAnnotation != null) {
                    String jpql = queryAnnotation.value();
                    
                    // Extract WHERE columns
                    List<String> whereColumns = extractWhereColumns(jpql);
                    recommendations.computeIfAbsent("WHERE", k -> new ArrayList<>())
                        .addAll(whereColumns);
                    
                    // Extract JOIN columns
                    List<String> joinColumns = extractJoinColumns(jpql);
                    recommendations.computeIfAbsent("JOIN", k -> new ArrayList<>())
                        .addAll(joinColumns);
                    
                    // Extract ORDER BY columns
                    List<String> sortColumns = extractOrderByColumns(jpql);
                    recommendations.computeIfAbsent("ORDER_BY", k -> new ArrayList<>())
                        .addAll(sortColumns);
                }
            }
            
            // Analyze JpaRepository method names
            for (Method method : repoClass.getDeclaredMethods()) {
                String methodName = method.getName();
                if (methodName.startsWith("findBy") || methodName.startsWith("countBy")) {
                    String columnName = extractColumnFromMethodName(methodName);
                    recommendations.computeIfAbsent("DERIVED_QUERY", k -> new ArrayList<>())
                        .add(columnName);
                }
            }
        }
        
        return recommendations;
    }
    
    private List<String> extractWhereColumns(String jpql) {
        List<String> columns = new ArrayList<>();
        Pattern pattern = Pattern.compile("WHERE\\s+(\\w+\\.\\w+)\\s*=", 
                                        Pattern.CASE_INSENSITIVE);
        Matcher matcher = pattern.matcher(jpql);
        while (matcher.find()) {
            columns.add(matcher.group(1));
        }
        return columns;
    }
```

```java
    @Autowired
    private JdbcTemplate jdbcTemplate;
    
    public List<IndexRecommendation> generateRecommendations() {
        List<IndexRecommendation> recommendations = new ArrayList<>();
        
        // 1. Foreign keys without indexes
        recommendations.addAll(findMissingForeignKeyIndexes());
        
        // 2. High-cardinality columns in WHERE clauses
        recommendations.addAll(findHighCardinalityWhereColumns());
        
        // 3. Columns used in JOINs
        recommendations.addAll(findJoinColumns());
        
        // 4. Columns used in ORDER BY with LIMIT
        recommendations.addAll(findSortColumns());
        
        // 5. Composite indexes for common query patterns
        recommendations.addAll(findCompositeIndexOpportunities());
        
        return prioritizeRecommendations(recommendations);
    }
    
    private List<IndexRecommendation> findMissingForeignKeyIndexes() {
        String sql = """
            SELECT
                tc.table_name,
                kcu.column_name,
                ccu.table_name AS foreign_table_name
            FROM information_schema.table_constraints AS tc
            JOIN information_schema.key_column_usage AS kcu
                ON tc.constraint_name = kcu.constraint_name
            JOIN information_schema.constraint_column_usage AS ccu
                ON ccu.constraint_name = tc.constraint_name
            WHERE tc.constraint_type = 'FOREIGN KEY'
              AND NOT EXISTS (
                  SELECT 1
                  FROM pg_indexes
                  WHERE tablename = tc.table_name
                    AND indexdef LIKE '%' || kcu.column_name || '%'
              )
            """;
        
        return jdbcTemplate.query(sql, (rs, rowNum) -> 
            IndexRecommendation.builder()
                .table(rs.getString("table_name"))
                .columns(List.of(rs.getString("column_name")))
                .type("B-tree")
                .priority(IndexPriority.HIGH)
                .reason("Missing index on foreign key - will slow JOINs")
                .estimatedImpact("High - foreign keys are always used in JOINs")
                .sql(String.format("CREATE INDEX idx_%s_%s ON %s(%s);",
                    rs.getString("table_name"),
                    rs.getString("column_name"),
                    rs.getString("table_name"),
                    rs.getString("column_name")))
                .build()
        );
    }
    
    private List<IndexRecommendation> findHighCardinalityWhereColumns() {
        String sql = """
            SELECT 
                query,
                calls,
                mean_exec_time,
                regexp_matches(query, 'FROM\s+(\w+)', 'i') AS table_name,
                regexp_matches(query, 'WHERE\s+(\w+)\s*=', 'gi') AS column_name
            FROM pg_stat_statements
            WHERE mean_exec_time > 50
              AND calls > 100
              AND query LIKE '%WHERE%=%'
            """;
        
        List<IndexRecommendation> recommendations = new ArrayList<>();
        
        jdbcTemplate.query(sql, rs -> {
            String table = rs.getString("table_name");
            String column = rs.getString("column_name");
            
            // Check cardinality
            double cardinality = checkCardinality(table, column);
            
            if (cardinality > 0.1) {  // > 10% distinct values
                recommendations.add(
                    IndexRecommendation.builder()
                        .table(table)
                        .columns(List.of(column))
                        .type("B-tree")
                        .priority(IndexPriority.MEDIUM)
                        .reason(String.format("High-cardinality column (%.1f%%) used in WHERE clause", cardinality * 100))
                        .estimatedImpact(String.format("%.0fms * %d calls = %.0fs total time saved",
                            rs.getDouble("mean_exec_time"),
                            rs.getLong("calls"),
                            rs.getDouble("mean_exec_time") * rs.getLong("calls") / 1000))
                        .sql(String.format("CREATE INDEX idx_%s_%s ON %s(%s);",
                            table, column, table, column))
                        .build()
                );
            }
        });
        
        return recommendations;
    }
    
    private double checkCardinality(String table, String column) {
        String sql = String.format(
            "SELECT COUNT(DISTINCT %s)::float / COUNT(*) FROM %s",
            column, table
        );
        return jdbcTemplate.queryForObject(sql, Double.class);
    }
```

```java
public class IndexPrioritizer {
    
    public List<IndexRecommendation> prioritize(List<IndexRecommendation> recommendations) {
        // Calculate score for each recommendation
        return recommendations.stream()
            .map(rec -> {
                int score = calculateScore(rec);
                rec.setScore(score);
                return rec;
            })
            .sorted(Comparator.comparingInt(IndexRecommendation::getScore).reversed())
            .collect(Collectors.toList());
    }
    
    private int calculateScore(IndexRecommendation rec) {
        int score = 0;
        
        // Factor 1: Query frequency
        if (rec.getCalls() > 1000) score += 50;
        else if (rec.getCalls() > 100) score += 30;
        else if (rec.getCalls() > 10) score += 10;
        
        // Factor 2: Query slowness
        if (rec.getMeanTime() > 100) score += 40;
        else if (rec.getMeanTime() > 50) score += 25;
        else if (rec.getMeanTime() > 10) score += 10;
        
        // Factor 3: Total impact (frequency * slowness)
        double totalImpact = rec.getCalls() * rec.getMeanTime();
        if (totalImpact > 100000) score += 60;  // > 100 seconds total
        else if (totalImpact > 10000) score += 35;  // > 10 seconds
        else if (totalImpact > 1000) score += 15;  // > 1 second
        
        // Factor 4: Column cardinality
        if (rec.getCardinality() > 0.5) score += 30;  // > 50% distinct
        else if (rec.getCardinality() > 0.1) score += 15;  // > 10% distinct
        
        // Factor 5: Index type
        if (rec.getType().equals("Foreign Key")) score += 40;  // Always create FK indexes
        
        // Factor 6: Table size
        if (rec.getTableRows() > 1000000) score += 20;  // Large table benefits more
        else if (rec.getTableRows() < 1000) score -= 20;  // Small table doesn't need index
        
        return score;
    }
}
```

### Q2: "Explain the trade-off between index count and write performance."

**Answer**: Each index adds overhead to INSERT/UPDATE/DELETE (3-5ms per index). 10 indexes = 30-50ms extra write time. Reads benefit (~10-100x faster), writes suffer (~5-10% slower per index). Optimal: 3-7 indexes per table. Priority: foreign keys, high-selectivity WHERE columns, frequent ORDER BY. Drop unused indexes (idx_scan = 0).

```java
public class IndexOverheadCalculator {
    
    public IndexImpact calculateImpact(TableStats stats) {
        // Model parameters
        double baseWriteTime = 1.0;  // ms per write without indexes
        double indexOverhead = 0.005;  // ms per index per write
        double readSpeedup = 50.0;  // average speedup with proper index
        
        // Calculate write impact
        double currentWriteTime = baseWriteTime + (stats.getIndexCount() * indexOverhead);
        double writesPerSec = stats.getWritesPerDay() / 86400.0;
        double dailyWriteTime = writesPerSec * currentWriteTime * 86400;
        
        // Calculate read impact
        double baseReadTime = 50.0;  // ms per read without index
        double indexedReadTime = baseReadTime / readSpeedup;
        double readsPerSec = stats.getReadsPerDay() / 86400.0;
        double dailyReadTime = readsPerSec * indexedReadTime * 86400;
        
        // Calculate optimal index count
        int optimalIndexes = calculateOptimalIndexCount(stats);
        
        return IndexImpact.builder()
            .currentIndexes(stats.getIndexCount())
            .optimalIndexes(optimalIndexes)
            .writeOverhead(stats.getIndexCount() * indexOverhead)
            .readImprovement(readSpeedup)
            .dailyWriteTime(dailyWriteTime)
            .dailyReadTime(dailyReadTime)
            .recommendation(generateRecommendation(stats, optimalIndexes))
            .build();
    }
    
    private int calculateOptimalIndexCount(TableStats stats) {
        double readWriteRatio = (double) stats.getReadsPerDay() / stats.getWritesPerDay();
        
        // Optimal index count based on read/write ratio
        if (readWriteRatio > 100) {
            return 10;  // Read-heavy: more indexes acceptable
        } else if (readWriteRatio > 10) {
            return 7;   // Balanced: moderate indexes
        } else if (readWriteRatio > 1) {
            return 5;   // Slight read preference
        } else {
            return 3;   // Write-heavy: minimal indexes
        }
    }
}
```

**Decision Framework**

```java
public class IndexDecisionMaker {
    
    public Decision shouldCreateIndex(IndexProposal proposal) {
        // Gather metrics
        TableMetrics metrics = metricsService.getTableMetrics(proposal.getTable());
        
        // Calculate costs
        double writeCost = calculateWriteCost(proposal, metrics);
        double readBenefit = calculateReadBenefit(proposal, metrics);
        
        // Calculate ROI
        double roi = (readBenefit - writeCost) / writeCost;
        
        if (roi > 10) {
            return Decision.approve("High ROI: " + String.format("%.0f%%", roi * 100));
        } else if (roi > 2) {
            return Decision.approve("Moderate ROI: " + String.format("%.0f%%", roi * 100));
        } else if (roi > 0) {
            return Decision.conditional("Low ROI: " + String.format("%.0f%%", roi * 100), 
                                      "Consider if query is critical");
        } else {
            return Decision.reject("Negative ROI: " + String.format("%.0f%%", roi * 100));
        }
    }
    
    private double calculateWriteCost(IndexProposal proposal, TableMetrics metrics) {
        // Cost = writes per day * overhead per write
        double overhead = 0.005;  // 5ms per write per index
        return metrics.getWritesPerDay() * overhead;
    }
    
    private double calculateReadBenefit(IndexProposal proposal, TableMetrics metrics) {
        // Benefit = reads per day * time saved per read
        double currentTime = proposal.getCurrentQueryTime();  // ms
        double expectedTime = proposal.getExpectedQueryTime();  // ms
        double savings = currentTime - expectedTime;
        
        return metrics.getReadsPerDay() * savings / 1000;  // Convert to seconds
    }
}
```

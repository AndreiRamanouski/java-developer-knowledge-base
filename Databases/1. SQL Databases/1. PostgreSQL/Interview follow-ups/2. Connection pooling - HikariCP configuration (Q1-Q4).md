### Q1: "How would you diagnose a sudden increase in connection pool wait times?"

**Answer**: Use a systematic approach: check pool metrics, identify slow queries, analyze database load, review application changes, and examine infrastructure issues.

**Step 1: Check Pool Metrics**

```java
@Component
public class ConnectionPoolDiagnostics {
    
    @Autowired
    private HikariDataSource dataSource;
    
    @Autowired
    private MeterRegistry meterRegistry;
    
    public DiagnosticReport diagnoseWaitTimeIncrease() {
        HikariPoolMXBean pool = dataSource.getHikariPoolMXBean();
        DiagnosticReport report = new DiagnosticReport();
        
        // Step 1: Collect current metrics
        int total = pool.getTotalConnections();
        int active = pool.getActiveConnections();
        int idle = pool.getIdleConnections();
        int waiting = pool.getThreadsAwaitingConnection();
        
        report.setTotalConnections(total);
        report.setActiveConnections(active);
        report.setIdleConnections(idle);
        report.setThreadsWaiting(waiting);
        
        // Step 2: Calculate utilization
        double utilization = (double) active / total * 100;
        report.setUtilization(utilization);
        
        // Step 3: Check historical metrics
        Timer acquireTimer = meterRegistry.find("hikaricp.connections.acquire")
            .timer();
        
        if (acquireTimer != null) {
            double avgAcquireTime = acquireTimer.mean(TimeUnit.MILLISECONDS);
            double maxAcquireTime = acquireTimer.max(TimeUnit.MILLISECONDS);
            
            report.setAvgAcquireTime(avgAcquireTime);
            report.setMaxAcquireTime(maxAcquireTime);
            
            // Red flag: Average wait time > 1 second
            if (avgAcquireTime > 1000) {
                report.addIssue("High average connection acquire time: " + 
                               avgAcquireTime + "ms");
            }
        }
        
        // Step 4: Identify root cause
        if (utilization > 90 && waiting > 0) {
            report.setProblemType("POOL_EXHAUSTION");
            report.addIssue("Pool is exhausted: " + utilization + "% utilized");
            report.addRecommendation("Check for connection leaks");
            report.addRecommendation("Identify slow queries");
            report.addRecommendation("Consider increasing pool size");
        }
        
        if (utilization < 50 && waiting > 0) {
            report.setProblemType("SLOW_QUERIES");
            report.addIssue("Low utilization but threads waiting");
            report.addRecommendation("Connections are being held too long");
            report.addRecommendation("Check for long-running queries");
        }
        
        return report;
    }
}

@Data
class DiagnosticReport {
    private int totalConnections;
    private int activeConnections;
    private int idleConnections;
    private int threadsWaiting;
    private double utilization;
    private double avgAcquireTime;
    private double maxAcquireTime;
    private String problemType;
    private List<String> issues = new ArrayList<>();
    private List<String> recommendations = new ArrayList<>();
    
    public void addIssue(String issue) {
        issues.add(issue);
    }
    
    public void addRecommendation(String recommendation) {
        recommendations.add(recommendation);
    }
}
```

**Step 2: Identify Slow Queries**

```java
@Component
public class SlowQueryAnalyzer {
    
    @Autowired
    private DataSource dataSource;
    
    public List<SlowQueryInfo> findSlowQueries() {
        // PostgreSQL: Use pg_stat_statements
        String sql = """
            SELECT 
                query,
                calls,
                total_exec_time,
                mean_exec_time,
                max_exec_time,
                stddev_exec_time
            FROM pg_stat_statements
            WHERE mean_exec_time > 100  -- Queries slower than 100ms
            ORDER BY mean_exec_time DESC
            LIMIT 20
            """;
        
        try (Connection conn = dataSource.getConnection();
             PreparedStatement ps = conn.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            
            List<SlowQueryInfo> slowQueries = new ArrayList<>();
            
            while (rs.next()) {
                SlowQueryInfo query = SlowQueryInfo.builder()
                    .query(rs.getString("query"))
                    .calls(rs.getLong("calls"))
                    .totalTime(rs.getDouble("total_exec_time"))
                    .avgTime(rs.getDouble("mean_exec_time"))
                    .maxTime(rs.getDouble("max_exec_time"))
                    .stddev(rs.getDouble("stddev_exec_time"))
                    .build();
                
                slowQueries.add(query);
            }
            
            return slowQueries;
            
        } catch (SQLException e) {
            log.error("Failed to retrieve slow queries", e);
            return Collections.emptyList();
        }
    }
    
    public List<BlockedQuery> findBlockedQueries() {
        // Find queries blocked by locks
        String sql = """
            SELECT
                blocked_locks.pid AS blocked_pid,
                blocked_activity.usename AS blocked_user,
                blocking_locks.pid AS blocking_pid,
                blocking_activity.usename AS blocking_user,
                blocked_activity.query AS blocked_statement,
                blocking_activity.query AS blocking_statement,
                blocked_activity.application_name
            FROM pg_catalog.pg_locks blocked_locks
            JOIN pg_catalog.pg_stat_activity blocked_activity 
                ON blocked_activity.pid = blocked_locks.pid
            JOIN pg_catalog.pg_locks blocking_locks 
                ON blocking_locks.locktype = blocked_locks.locktype
            JOIN pg_catalog.pg_stat_activity blocking_activity 
                ON blocking_activity.pid = blocking_locks.pid
            WHERE NOT blocked_locks.granted;
            """;
        
        try (Connection conn = dataSource.getConnection();
             PreparedStatement ps = conn.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            
            List<BlockedQuery> blockedQueries = new ArrayList<>();
            
            while (rs.next()) {
                BlockedQuery query = BlockedQuery.builder()
                    .blockedPid(rs.getInt("blocked_pid"))
                    .blockingPid(rs.getInt("blocking_pid"))
                    .blockedQuery(rs.getString("blocked_statement"))
                    .blockingQuery(rs.getString("blocking_statement"))
                    .applicationName(rs.getString("application_name"))
                    .build();
                
                blockedQueries.add(query);
            }
            
            return blockedQueries;
            
        } catch (SQLException e) {
            log.error("Failed to retrieve blocked queries", e);
            return Collections.emptyList();
        }
    }
}

}

    private int blockingPid;
    private String blockedQuery;
    private String blockingQuery;
    private String applicationName;
```

**Step 3: Compare with Historical Baseline**

```java
@Service
public class BaselineComparisonService {
    
    @Autowired
    private MeterRegistry meterRegistry;
    
    private final Map<String, Double> baseline = new ConcurrentHashMap<>();
    
    @Scheduled(fixedDelay = 300000) // Every 5 minutes
    public void recordBaseline() {
        // Record current metrics as baseline
        Timer acquireTimer = meterRegistry.find("hikaricp.connections.acquire")
            .timer();
        
        if (acquireTimer != null) {
            baseline.put("avg_acquire_time", acquireTimer.mean(TimeUnit.MILLISECONDS));
            baseline.put("max_acquire_time", acquireTimer.max(TimeUnit.MILLISECONDS));
        }
        
        Gauge activeGauge = meterRegistry.find("hikaricp.connections.active")
            .gauge();
        
        if (activeGauge != null) {
            baseline.put("active_connections", activeGauge.value());
        }
    }
    
    public DeviationReport checkDeviation() {
        DeviationReport report = new DeviationReport();
        
        Timer currentAcquire = meterRegistry.find("hikaricp.connections.acquire")
            .timer();
        
        if (currentAcquire != null && baseline.containsKey("avg_acquire_time")) {
            double currentAvg = currentAcquire.mean(TimeUnit.MILLISECONDS);
            double baselineAvg = baseline.get("avg_acquire_time");
            double deviation = ((currentAvg - baselineAvg) / baselineAvg) * 100;
            
            report.setMetric("avg_acquire_time");
            report.setCurrentValue(currentAvg);
            report.setBaselineValue(baselineAvg);
            report.setDeviationPercent(deviation);
            
            // Alert if deviation > 50%
            if (deviation > 50) {
                report.setAlert(true);
                report.setMessage(String.format(
                    "Connection acquire time increased by %.2f%% (%.2fms → %.2fms)",
                    deviation, baselineAvg, currentAvg
                ));
            }
        }
        
        return report;
    }
}

@Data
class DeviationReport {
    private String metric;
    private double currentValue;
    private double baselineValue;
    private double deviationPercent;
    private boolean alert;
    private String message;
}
```

### Q2: "What's the impact of setting minimum-idle equal to maximum-pool-size?"

**Answer**: Setting `minimum-idle = maximum-pool-size` eliminates dynamic scaling, wastes resources during low traffic, increases startup time, and removes HikariCP's automatic optimization benefits. It should only be used for predictable, constant load scenarios.

**Impact Analysis**

```yaml
# Configuration 1: Dynamic scaling (RECOMMENDED)
spring:
  datasource:
    hikari:
      maximum-pool-size: 20
      minimum-idle: 5  # 25% of max

# Configuration 2: Fixed pool (NOT RECOMMENDED)
spring:
  datasource:
    hikari:
      maximum-pool-size: 20
      minimum-idle: 20  # Same as max
```

**Impact 1: Startup Time**

### Q3: "How do you handle connection pooling in a multi-tenant application?"

**Answer**: Use separate connection pools per tenant for isolation, or a single shared pool with tenant context for efficiency. Choice depends on tenant isolation requirements, number of tenants, and resource constraints.

**Strategy 1: Separate Pool Per Tenant (Strong Isolation)**

```java
@Configuration
public class MultiTenantDataSourceConfig {
    
    private final Map<String, HikariDataSource> tenantDataSources = 
        new ConcurrentHashMap<>();
    
    @Bean
    public DataSource dataSource() {
        return new TenantRoutingDataSource();
    }
    
    public HikariDataSource createTenantDataSource(String tenantId) {
        return tenantDataSources.computeIfAbsent(tenantId, id -> {
            HikariConfig config = new HikariConfig();
            
            // Tenant-specific database
            config.setJdbcUrl(String.format(
                "jdbc:postgresql://localhost:5432/tenant_%s", tenantId
            ));
            config.setUsername("tenant_" + tenantId);
            config.setPassword(getTenantPassword(tenantId));
            
            // Smaller pool per tenant
            int maxPoolSize = 5;  // Instead of 20 for shared
            config.setMaximumPoolSize(maxPoolSize);
            config.setMinimumIdle(2);
            
            // Tenant-specific pool name
            config.setPoolName("HikariPool-Tenant-" + tenantId);
            
            // Shorter timeouts (resource protection)
            config.setConnectionTimeout(20000);
            config.setIdleTimeout(300000);
            config.setMaxLifetime(900000);
            
            log.info("Created connection pool for tenant: {}", tenantId);
            
            return new HikariDataSource(config);
        });
    }
    
    public void removeTenantDataSource(String tenantId) {
        HikariDataSource dataSource = tenantDataSources.remove(tenantId);
        if (dataSource != null) {
            dataSource.close();
            log.info("Closed connection pool for tenant: {}", tenantId);
        }
    }
    
    private String getTenantPassword(String tenantId) {
        // Retrieve from secure vault
        return "password";
    }
}

// Dynamic routing data source
public class TenantRoutingDataSource extends AbstractRoutingDataSource {
    
    @Autowired
    private MultiTenantDataSourceConfig dataSourceConfig;
    
    @Override
    protected Object determineCurrentLookupKey() {
        return TenantContext.getCurrentTenant();
    }
    
    @Override
    protected DataSource determineTargetDataSource() {
        String tenantId = (String) determineCurrentLookupKey();
        
        if (tenantId == null) {
            throw new IllegalStateException("No tenant context set");
        }
        
        return dataSourceConfig.createTenantDataSource(tenantId);
    }
}

// Tenant context (thread-local)
public class TenantContext {
    
    private static final ThreadLocal<String> currentTenant = new ThreadLocal<>();
    
    public static void setCurrentTenant(String tenantId) {
        currentTenant.set(tenantId);
    }
    
    public static String getCurrentTenant() {
        return currentTenant.get();
    }
    
    public static void clear() {
        currentTenant.remove();
    }
}

// Interceptor to set tenant context
@Component
public class TenantInterceptor implements HandlerInterceptor {
    
    @Override
    public boolean preHandle(HttpServletRequest request, 
                            HttpServletResponse response, 
                            Object handler) {
        
        // Extract tenant from header, subdomain, or path
        String tenantId = extractTenantId(request);
        
        if (tenantId != null) {
            TenantContext.setCurrentTenant(tenantId);
        }
        
        return true;
    }
    
    @Override
    public void afterCompletion(HttpServletRequest request, 
                               HttpServletResponse response, 
                               Object handler, 
                               Exception ex) {
        TenantContext.clear();
    }
    
    private String extractTenantId(HttpServletRequest request) {
        // Option 1: Header
        String tenantId = request.getHeader("X-Tenant-ID");
        if (tenantId != null) return tenantId;
        
        // Option 2: Subdomain
        String host = request.getServerName();
        if (host.contains(".")) {
            return host.substring(0, host.indexOf("."));
        }
        
        // Option 3: Path
        String path = request.getRequestURI();
        if (path.startsWith("/tenant/")) {
            return path.split("/")[2];
        }
        
        return null;
    }
}
```

**Pros and Cons of Separate Pools**

```java
// ✅ Pros:
// - Strong isolation (tenant cannot affect another's connections)
// - Per-tenant monitoring and metrics
// - Per-tenant database (data isolation)
// - Individual tenant can be throttled/suspended
// - Easier to debug tenant-specific issues

// ❌ Cons:
// - High memory usage (each pool uses ~50-200MB)
//   100 tenants * 5 connections * 10MB = 5GB minimum
```

**Per-Tenant Connection Limits**

```java
@Component
public class TenantConnectionThrottler {
    
    private final Map<String, Semaphore> tenantLimits = new ConcurrentHashMap<>();
    
    // Limit each tenant to maximum 5 concurrent connections
    private static final int MAX_CONNECTIONS_PER_TENANT = 5;
    
    public Connection acquireConnectionForTenant(DataSource dataSource, 
                                                 String tenantId) 
            throws SQLException, InterruptedException {
        
        Semaphore limit = tenantLimits.computeIfAbsent(tenantId, 
            id -> new Semaphore(MAX_CONNECTIONS_PER_TENANT));
        
        // Block if tenant already has MAX_CONNECTIONS_PER_TENANT active
        limit.acquire();
        
        try {
            return dataSource.getConnection();
        } catch (SQLException e) {
            limit.release();
            throw e;
        }
    }
    
    public void releaseConnection(Connection connection, String tenantId) 
            throws SQLException {
        try {
            connection.close();
        } finally {
            Semaphore limit = tenantLimits.get(tenantId);
            if (limit != null) {
                limit.release();
            }
        }
    }
}

// Usage in repository
@Repository

    private DataSource dataSource;

        String tenantId = TenantContext.getCurrentTenant();

            Connection conn = throttler.acquireConnectionForTenant(
                dataSource, tenantId
            );

                    "SELECT * FROM orders WHERE tenant_id = ?")) {
                ps.setString(1, tenantId);
                ResultSet rs = ps.executeQuery();
                // Process results...
                return orders;
            } finally {
                throttler.releaseConnection(conn, tenantId);
            }
```

```java
            ))
            .limit(limit)
            .collect(Collectors.toMap(
                Map.Entry::getKey,
                e -> e.getValue().get()
            ));
    }
```

### Q4: "Explain the difference between idle-timeout and max-lifetime."

**Answer**: `idle-timeout` removes connections that sit unused, while `max-lifetime` closes connections after a total age regardless of usage. Both prevent stale connections but serve different purposes: idle-timeout conserves resources, max-lifetime prevents connection corruption and respects database timeouts.

**idle-timeout: Removes Unused Connections**

```yaml
spring:
  datasource:
    hikari:
      idle-timeout: 600000  # 10 minutes (default)
      # Connections that sit idle for 10+ minutes are removed
```

```java
// How idle-timeout works:

public class IdleTimeoutExplained {
    
    public void explainIdleTimeout() {
        // Timeline for a connection:
        
        // T=0: Connection created
        // T=1min: Used for query
        // T=2min: Returned to pool (starts idle timer)
        // T=12min: Still idle (exceeded 10-minute idle-timeout)
        // → Connection closed and removed from pool
        
        // Timeline for an active connection:
        // T=0: Connection created
        // T=5min: Used for query
        // T=6min: Returned to pool (starts idle timer)
        // T=10min: Used again for query (idle timer resets)
        // T=11min: Returned to pool (starts new idle timer)
        // T=21min: Still idle (exceeded 10-minute idle-timeout)
        // → Connection closed and removed from pool
        
        // Key point: Timer only counts idle time, not total time
    }
}
```

**max-lifetime: Closes Old Connections**

```yaml
spring:
  datasource:
    hikari:
      max-lifetime: 1800000  # 30 minutes (default)
      # Connections older than 30 minutes are closed (even if actively used)
```

```java
// How max-lifetime works:

public class MaxLifetimeExplained {
    
    public void explainMaxLifetime() {
        // Timeline for a connection:
        
        // T=0: Connection created (max-lifetime timer starts)
        // T=5min: Used for query
        // T=10min: Used for query
        // T=15min: Used for query
        // T=30min: Connection reaches max-lifetime
        // → Connection marked for closing
        // → Next time it's returned to pool, it's closed
        
        // Key point: Timer counts total time since creation,
        //           regardless of usage pattern
    }
}
```

```java
            // Will get a NEW connection because old one expired
        }

        // Output: Total=2 (old connection replaced with new one)

    }
```

**Database Timeout Coordination**

```java
// Critical: max-lifetime must be < database timeout

// PostgreSQL:
// SHOW idle_in_transaction_session_timeout;
// Default: 0 (disabled)
// Recommended: 1800000 (30 minutes)

// MySQL:
// SHOW VARIABLES LIKE 'wait_timeout';
// Default: 28800 seconds (8 hours)
// Recommended: 3600 seconds (1 hour)

// HikariCP configuration:
// Rule: max-lifetime should be 30-60 seconds LESS than database timeout
// Why: Gives time to close connection gracefully before database kills it

@Configuration
public class TimeoutCoordinationConfig {
    
    @Bean
    public DataSource dataSource() {
        HikariConfig config = new HikariConfig();
        
        // Database wait_timeout = 1800 seconds (30 minutes)
        // HikariCP max-lifetime = 1740 seconds (29 minutes)
        // Safety margin: 60 seconds
        
        config.setMaxLifetime(1740000);  // 29 minutes
        
        // idle-timeout should be < max-lifetime
        config.setIdleTimeout(600000);   // 10 minutes
        
        return new HikariDataSource(config);
    }
}
```
